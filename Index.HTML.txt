    <script>
        // Session and State Variables
        let totalInvoicesGenerated = 0;
        let totalRevenueGenerated = 0;
        let currentInvoiceActive = false;
        let currentInvoiceTotal = 0;

        function generateRunnersHubInvoice() {
            let client = document.getElementById('clientName').value || "Valued Client";
            let item1 = document.getElementById('itemName1').value || "Concierge Service";
            let price1 = parseFloat(document.getElementById('itemPrice1').value) || 0;
            let item2 = document.getElementById('itemName2').value || "";
            let price2 = parseFloat(document.getElementById('itemPrice2').value) || 0;
            
            let customName = document.getElementById('customName').value.trim();
            let customPrice = parseFloat(document.getElementById('customPrice').value) || 0;

            let delivery = parseFloat(document.getElementById('deliveryFee').value) || 0;
            let discount = parseFloat(document.getElementById('discountAmt').value) || 0;

            if (customPrice > 0 && customName === "") {
                alert("Action Required: You entered a price for a Custom Part/Service, but left the description blank. Please specify what it is.");
                document.getElementById('customName').focus(); 
                return; 
            }

            let subtotal = price1 + price2 + customPrice + delivery;
            if (discount > subtotal) discount = subtotal; 
            
            let taxableAmount = subtotal - discount;
            let vat = taxableAmount * 0.15;
            let grandTotal = taxableAmount + vat;

            document.getElementById('resClient').innerText = client;
            
            if (price1 > 0) {
                document.getElementById('lblItem1').innerText = item1;
                document.getElementById('resPrice1').innerText = "Br " + price1.toFixed(2);
                document.getElementById('resRow1').style.display = 'flex';
            } else {
                document.getElementById('resRow1').style.display = 'none';
            }
            
            if (price2 > 0 && item2 !== "") {
                document.getElementById('lblItem2').innerText = item2;
                document.getElementById('resPrice2').innerText = "Br " + price2.toFixed(2);
                document.getElementById('resRow2').style.display = 'flex';
            } else {
                document.getElementById('resRow2').style.display = 'none';
            }

            if (customPrice > 0) {
                document.getElementById('lblCustom').innerText = customName;
                document.getElementById('resPriceCustom').innerText = "Br " + customPrice.toFixed(2);
                document.getElementById('resRowCustom').style.display = 'flex';
            } else {
                document.getElementById('resRowCustom').style.display = 'none';
            }

            if (delivery > 0) {
                document.getElementById('resDelivery').innerText = "Br " + delivery.toFixed(2);
                document.getElementById('resRowDelivery').style.display = 'flex';
            } else {
                document.getElementById('resRowDelivery').style.display = 'none';
            }

            document.getElementById('resSubtotal').innerText = "Br " + subtotal.toFixed(2);
            
            if (discount > 0) {
                document.getElementById('resDiscount').innerText = "- Br " + discount.toFixed(2);
                document.getElementById('resRowDiscount').style.display = 'flex';
            } else {
                document.getElementById('resRowDiscount').style.display = 'none';
            }

            document.getElementById('resVat').innerText = "Br " + vat.toFixed(2);
            document.getElementById('resTotal').innerText = "Br " + grandTotal.toFixed(2);

            // Update Session Dashboard Summary
            // If there is already an active invoice on screen, we don't double count it.
            // We just add the new one.
            totalInvoicesGenerated++;
            totalRevenueGenerated += grandTotal;
            
            document.getElementById('totalInvoices').innerText = totalInvoicesGenerated;
            document.getElementById('totalRevenue').innerText = "Br " + totalRevenueGenerated.toFixed(2);

            // Store current invoice state for potential voiding
            currentInvoiceTotal = grandTotal;
            currentInvoiceActive = true;

            // Show Invoice Box
            document.getElementById('invoiceDisplay').style.display = 'block';
        }

        function voidCurrentInvoice() {
            if (!currentInvoiceActive) return;

            // Prompt for Master Password
            let passwordAttempt = prompt("Manager authorization required to delete this invoice.\n\nEnter Master Password:");
            
            if (passwordAttempt === "RunnersHUB123") {
                // Correct Password: Undo the session statistics
                totalInvoicesGenerated--;
                totalRevenueGenerated -= currentInvoiceTotal;

                // Safety net to prevent negative stats
                if (totalInvoicesGenerated < 0) totalInvoicesGenerated = 0;
                if (totalRevenueGenerated < 0) totalRevenueGenerated = 0;

                // Update Dashboard UI
                document.getElementById('totalInvoices').innerText = totalInvoicesGenerated;
                document.getElementById('totalRevenue').innerText = "Br " + totalRevenueGenerated.toFixed(2);

                // Hide the invoice display and reset state
                document.getElementById('invoiceDisplay').style.display = 'none';
                currentInvoiceActive = false;
                currentInvoiceTotal = 0;

                alert("Success: Invoice deleted and session statistics reverted.");
            } else if (passwordAttempt !== null) {
                // Wrong Password (and user didn't hit 'Cancel')
                alert("Error: Incorrect Master Password. Invoice deletion denied.");
            }
        }
    </script>
